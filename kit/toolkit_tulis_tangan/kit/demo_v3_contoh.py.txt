ELS = []
SCENES = []
CUR = None
TOTAL = 190.1


def scene(t0, t1, wh=True):
    global CUR
    CUR = {'t0': t0, 't1': t1, 'els': []}
    SCENES.append(CUR)
    if wh:
        sfx(t0, 'whoosh', 0, 0.55)


def A(el):
    if el.out is None:
        el.out = CUR['t1']
    CUR['els'].append(el)
    ELS.append(el)
    el.register()
    return el


def DIM(t, *keep):
    for el in CUR['els']:
        if el.dim is None and el.t0 < t and el not in keep:
            el.dim = t


def head(t, n, tt=None):
    A(Ticks(t, n, total=4, x0=240, x1=480))
    A(Txt(tt or t + 0.15, f'Kehormatan ke-{n}', 360, 262, 30))


def rasul(t, cx, cy, size=26, color=NAVY, extra=''):
    txt = 'Rasulullah' + extra
    w1 = tsprite(txt, 'kb', size, color).w
    w2 = tsprite('ﷺ', 'ar', int(size * 1.25), color).w
    left = cx - (w1 + 4 + w2) / 2
    A(Txt(t, txt, left, cy, size, color=color, ax='l'))
    A(Txt(t + 0.3, 'ﷺ', left + w1 + 4, cy + 2, int(size * 1.25), fn='ar', color=color, ax='l', mode='pop'))
