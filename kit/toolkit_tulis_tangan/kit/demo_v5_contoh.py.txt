from engine import *

ELS = []; SCENES = []; CUR = None
TOTAL = 44.0
AR = dict(fn='ar')


def scene(t0, t1, wh=True):
    global CUR
    CUR = {'t0': t0, 't1': t1, 'els': []}; SCENES.append(CUR)
    if wh:
        sfx(t0, 'whoosh', 0, 0.55)


def A(el):
    if el.out is None:
        el.out = CUR['t1']
    CUR['els'].append(el); ELS.append(el); el.register(); return el


def AA(els):
    return [A(e) for e in els]
