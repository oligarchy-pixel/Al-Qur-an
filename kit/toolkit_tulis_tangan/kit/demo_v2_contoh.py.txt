def scene(t0, t1, wh=True):
    global CUR
    CUR = {'t0': t0, 't1': t1, 'els': []}
    SCENES.append(CUR)
    if wh:
        sfx(t0, 'whoosh', 0, 0.55)


def A(el):
    if el.out is None:
        el.out = CUR['t1']
    CUR['els'].append(el)
    ELS.append(el)
    el.register()
    return el


def DIM(t, *keep):
    for el in CUR['els']:
        if el.dim is None and el.t0 < t and el not in keep:
            el.dim = t

def rasul(t, cx, cy, size=26, color=NAVY, extra=''):
    txt = 'Rasulullah' + extra
    w1 = tsprite(txt, 'kb', size, color).w
    w2 = tsprite('ﷺ', 'ar', int(size * 1.25), color).w
    left = cx - (w1 + 4 + w2) / 2
    A(Txt(t, txt, left, cy, size, color=color, ax='l'))
    A(Txt(t + 0.3, 'ﷺ', left + w1 + 4, cy + 2, int(size * 1.25), fn='ar', color=color, ax='l', mode='pop'))

def ar_word_span(text, word, cx, size):
    f = font('ar', size); full = f.getlength(text, direction='rtl', language='ar')
    ww = f.getlength(word, direction='rtl', language='ar')
    left = cx - full / 2
    idx = text.split(' ').index(word)
    after = ' '.join(text.split(' ')[idx + 1:])
    off = f.getlength(after, direction='rtl', language='ar') + (f.getlength(' ') if after else 0)
    return left + off, left + off + ww
def lat_span(text, sub, cx, size):
    f = font('kr', size); full = f.getlength(text); i = text.index(sub)
    x0 = cx - full / 2 + f.getlength(text[:i]); return x0, x0 + f.getlength(sub)
