import sys
from PIL import Image
fs = sys.argv[2:]; out = sys.argv[1]; w, h = 360, 640; cols = 5; rows = (len(fs) + cols - 1) // cols
S = Image.new('RGB', (w * cols, h * rows), 'white')
for i, f in enumerate(fs):
    S.paste(Image.open(f).resize((w, h)), ((i % cols) * w, (i // cols) * h))
S.save(out)
