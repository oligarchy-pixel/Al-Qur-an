ELS = []
SCENES = []

def scene(t0, t1, wh=True):
    global CUR
    CUR = {'t0': t0, 't1': t1, 'els': []}
    SCENES.append(CUR)
    if wh:
        sfx(t0, 'whoosh', 0, 0.55)


def A(el):
    if el.out is None:
        el.out = CUR['t1']
    CUR['els'].append(el)
    ELS.append(el)
    el.register()
    return el


def AA(els):
    return [A(e) for e in els]


def DIM(t, *keep):
    for el in CUR['els']:
        if el.dim is None and el.t0 < t and el not in keep:
            el.dim = t


def CLR(t):
    for el in CUR['els']:
        if el.out is None or el.out > t:
            el.out = t


def rasul(t, cx, cy, size=26, color=NAVY, pre=''):
    txt = pre + 'Rasulullah'
    w1 = tsprite(txt, 'kb', size, color).w
    w2 = tsprite('ﷺ', 'ar', int(size * 1.25), color).w
    left = cx - (w1 + 4 + w2) / 2
    A(Txt(t, txt, left, cy, size, color=color, ax='l'))
    A(Txt(t + 0.3, 'ﷺ', left + w1 + 4, cy + 2, int(size * 1.25), fn='ar', color=color, ax='l', mode='pop'))


class Num(El):
    """angka menghitung naik dengan pemisah titik ribuan"""

    def __init__(self, t0, cx, cy, n, suffix='', size=60, color=GREEN, dur=1.4, **kw):
        super().__init__(t0, **kw)
        self.cx, self.cy, self.n, self.suffix, self.size, self.color, self.dur = cx, cy, n, suffix, size, color, dur

    def register(self):
        sfx(self.t0, 'rise', self.dur, 0.6)
        for i in range(8):
            sfx(self.t0 + i * self.dur / 8, 'tick', 0, 0.3)

    def draw(self, fr, t):
        v = int(round(self.n * eo((t - self.t0) / self.dur)))
        s = f'{v:,}'.replace(',', '.')
        sp = tsprite(s + self.suffix, 'kb', self.size, self.color)
        paste(fr, sp.im, self.cx, self.cy, self.fade(t))
