def put(sp, cx, cy, rot=0):
    if rot: sp = sp.rotate(rot, Image.BICUBIC, expand=True)
    im.alpha_composite(sp, (int(cx - sp.width/2), int(cy - sp.height/2)))

def text(t, cx, cy, fn, size, col, rot=0):
    S = 2
    f = font(fn, size*S)
    kw = dict(direction='rtl', language='ar') if fn in ('ar','arb') else {}
    bb = f.getbbox(t, **kw)
    layer = Image.new('RGBA', (bb[2]-bb[0]+40*S, bb[3]-bb[1]+40*S), (0,0,0,0))
    ImageDraw.Draw(layer).text((20*S-bb[0], 20*S-bb[1]), t, font=f, fill=col+(255,), **kw)
    layer = layer.resize((layer.width//S, layer.height//S), Image.LANCZOS)
    put(layer, cx, cy, rot)

def sticky(t, cx, cy, size, rot, col=YEL, sub=None):
    S = 2; f = font('kb', size*S)
    w = int(f.getlength(t)/S + size*1.6); h = int(size*(2.9 if sub else 2.0))
    s = Image.new('RGBA', (w*S, h*S), col+(255,)); d = ImageDraw.Draw(s)
    d.rectangle((0,0,w*S,8*S), fill=tuple(c-18 for c in col)+(255,))
    text_on(d, (w/2, size*1.05 if sub else h/2+2), t, 'kb', size, (60,50,30,255))
    if sub: text_on(d, (w/2, h-size*0.75), sub, 'kr', int(size*0.62), (80,70,50,255))
    s = shadowed(s.resize((w, h), Image.LANCZOS), (4,6), 6, 80)
    put(s, cx, cy, rot)

def ellipse(cx, cy, rx, ry, col, wdt):
    S = 3
    L = Image.new('RGBA', (W*S, H*S), (0,0,0,0)); d = ImageDraw.Draw(L)
    pts = [(S*(cx + rx*(1+.04*math.sin(i/70*7))*math.cos(math.radians(-120+385*i/70))),
            S*(cy + ry*(1+.04*math.sin(i/70*7))*math.sin(math.radians(-120+385*i/70)))) for i in range(71)]
    d.line(pts, fill=col+(255,), width=wdt*S, joint='curve')
    im.alpha_composite(L.resize((W, H), Image.LANCZOS))
