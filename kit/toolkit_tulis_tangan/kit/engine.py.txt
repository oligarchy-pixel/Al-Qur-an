@functools.lru_cache(None)
def font(name, size):
    path = {'kb': 'Kalam-Bold.ttf', 'kr': 'Kalam-Regular.ttf', 'kl': 'Kalam-Light.ttf',
            'ar': 'Amiri-Regular.ttf', 'arb': 'Amiri-Bold.ttf'}[name]
    return ImageFont.truetype(FD + path, size, layout_engine=ImageFont.Layout.RAQM)


def is_ar(s):
    return any(('؀' <= c <= 'ۿ') or ('ﭐ' <= c <= '﻿') or ('ݐ' <= c <= 'ݿ') for c in s)


def clamp(x, a=0.0, b=1.0):
    return a if x < a else b if x > b else x


def eo(x):  # ease out cubic
    x = clamp(x)
    return 1 - (1 - x) ** 3


def eio(x):
    x = clamp(x)
    return 3 * x * x - 2 * x * x * x


def eback(x, s=1.9):
    x = clamp(x)
    x -= 1
    return x * x * ((s + 1) * x + s) + 1


def mix(c1, c2, a):
    return tuple(int(c1[i] * (1 - a) + c2[i] * a) for i in range(3))


# ---------------------------------------------------------------- background
def make_bg():
    yy, xx = np.mgrid[0:H, 0:W].astype(np.float32)
    base = np.zeros((H, W, 3), np.float32)
    base[:] = (240, 239, 231)
    grid = ((xx.astype(int) % 36) == 0) | ((yy.astype(int) % 36) == 0)
    base[grid] = (226, 225, 216)
    d = np.sqrt(((xx - W / 2) / (W * 0.62)) ** 2 + ((yy - H * 0.47) / (H * 0.58)) ** 2)
    v = np.clip((d - 0.45) / 0.75, 0, 1) ** 1.6
    base *= (1 - 0.085 * v)[..., None]
    rng = np.random.default_rng(3)
    base += rng.normal(0, 1.4, (H, W, 1))
    return Image.fromarray(np.clip(base, 0, 255).astype(np.uint8), 'RGB')

# ---------------------------------------------------------------- sprite helpers
_acache = {}


def with_alpha(sp, a):
    if a >= 0.995:
        return sp
    q = max(1, int(a * 24 + 0.5))
    key = (id(sp), q)
    r = _acache.get(key)
    if r is None:
        f = q / 24.0
        r = sp.copy()
        r.putalpha(sp.getchannel('A').point(lambda v: int(v * f)))
        _acache[key] = r
        if len(_acache) > 6000:
            _acache.clear()
    return r


def paste(frame, sp, cx, cy, a=1.0, s=1.0, rot=0.0):
    if a <= 0.01 or s <= 0.02:
        return
    if abs(s - 1) > 0.01:
        sp = sp.resize((max(1, int(sp.width * s)), max(1, int(sp.height * s))), Image.BILINEAR)
    if rot:
        sp = sp.rotate(rot, Image.BILINEAR, expand=True)
    sp = with_alpha(sp, a) if (abs(s - 1) <= 0.01 and not rot) else _alpha_now(sp, a)
    frame.paste(sp, (int(cx - sp.width / 2), int(cy - sp.height / 2)), sp)


def _alpha_now(sp, a):
    if a >= 0.995:
        return sp
    sp = sp.copy()
    sp.putalpha(sp.getchannel('A').point(lambda v: int(v * a)))
    return sp


def shadowed(im, off=(3, 4), blur=4, alpha=60):
    pad = blur * 2 + max(off)
    out = Image.new('RGBA', (im.width + 2 * pad, im.height + 2 * pad), (0, 0, 0, 0))
    sh = Image.new('RGBA', out.size, (0, 0, 0, 0))
    a = im.getchannel('A').point(lambda v: v * alpha // 255)
    blk = Image.new('RGBA', im.size, (40, 40, 60, 255))
    blk.putalpha(a)
    sh.paste(blk, (pad + off[0], pad + off[1]), blk)
    sh = sh.filter(ImageFilter.GaussianBlur(blur))
    out = Image.alpha_composite(sh, out)
    out.paste(im, (pad, pad), im)
    return out


# ---------------------------------------------------------------- text sprite
class TextSprite:
    def __init__(self, text, fn='kb', size=34, color=NAVY, maxw=640, align='center', lh=1.12):
        rtl = is_ar(text) and fn in ('ar', 'arb')
        f = font(fn, size)
        kw = dict(direction='rtl', language='ar') if rtl else {}
        lines = []
        for para in text.split('\n'):
            cur = ''
            for w in para.split(' '):
                test = (cur + ' ' + w).strip()
                if maxw and cur and f.getlength(test, **kw) > maxw:
                    lines.append(cur)
                    cur = w
                else:
                    cur = test
            lines.append(cur)
        asc, desc = f.getmetrics()
        lhp = int(size * (1.75 if rtl else 1.18) * lh)
        widths = [f.getlength(l, **kw) for l in lines]
        pad = int(size * 0.9)
        Wd = int(max(widths)) + 2 * pad
        Ht = lhp * (len(lines) - 1) + int(size * 2.4) + 2 * pad
        im = Image.new('RGBA', (Wd, Ht), (0, 0, 0, 0))
        d = ImageDraw.Draw(im)
        boxes = []
        for i, l in enumerate(lines):
            if align == 'center':
                x = (Wd - widths[i]) / 2
            elif align == 'left':
                x = pad
            else:
                x = Wd - pad - widths[i]
            y = pad + i * lhp
            d.text((x, y), l, font=f, fill=color + (255,), **kw)
            boxes.append([x, y, x + widths[i], y + lhp, len(l)])
        bb = im.getbbox() or (0, 0, 2, 2)
        m = 3
        bb = (max(0, bb[0] - m), max(0, bb[1] - m), min(Wd, bb[2] + m), min(Ht, bb[3] + m))
        self.im = im.crop(bb)
        # line bands (y ranges) relative to crop
        bands = []
        for i, (x0, y0, x1, y1, n) in enumerate(boxes):
            ya = 0 if i == 0 else int((y0 - bb[1]))
            yb = self.im.height if i == len(boxes) - 1 else int(boxes[i + 1][1] - bb[1])
            bands.append((int(x0 - bb[0]) - 3, ya, int(x1 - bb[0]) + 6, yb, n))
        self.bands = bands
        self.rtl = rtl
        self.nchar = sum(b[4] for b in bands)
        self.w, self.h = self.im.size

    def revealed(self, p):
        if p >= 1:
            return self.im
        out = Image.new('RGBA', self.im.size, (0, 0, 0, 0))
        tot = max(1, self.nchar)
        acc = 0
        for (x0, y0, x1, y1, n) in self.bands:
            lp = clamp((p * tot - acc) / max(1, n))
            acc += n
            if lp <= 0:
                break
            x0 = max(0, x0)
            x1 = min(self.w, x1)
            wpx = int((x1 - x0) * lp)
            if wpx <= 0:
                continue
            if self.rtl:
                box = (x1 - wpx, y0, x1, y1)
            else:
                box = (x0, y0, x0 + wpx, y1)
            out.paste(self.im.crop(box), box[:2])
        return out


@functools.lru_cache(maxsize=4000)
def tsprite(text, fn='kb', size=34, color=NAVY, maxw=640, align='center'):
    return TextSprite(text, fn, size, color, maxw, align)


# ---------------------------------------------------------------- elements
class El:
    z = 0

    def __init__(self, t0, dim=None, out=None, sfx_on=True):
        self.t0 = t0
        self.dim = dim
        self.out = out
        self.sfx_on = sfx_on

    def fade(self, t):
        a = 1.0
        if self.dim is not None and t > self.dim:
            a *= 1 - 0.72 * eo((t - self.dim) / 0.35)
        if self.out is not None and t > self.out - 0.3:
            a *= clamp((self.out - t) / 0.3)
        return a

    def active(self, t):
        return t >= self.t0 and (self.out is None or t < self.out)

    def register(self):
        pass


class Txt(El):
    def __init__(self, t0, text, cx, cy, size=34, fn='kb', color=NAVY, maxw=620, mode='write', dur=None,
                 align='center', ax='c', rot=0, **kw):
        super().__init__(t0, **kw)
        self.sp = tsprite(text, fn, size, color, maxw, align)
        self.cx, self.cy, self.mode, self.rot = cx, cy, mode, rot
        if ax == 'l':
            self.cx = cx + self.sp.w / 2
        elif ax == 'r':
            self.cx = cx - self.sp.w / 2
        n = self.sp.nchar
        self.dur = dur if dur is not None else clamp(n * (0.045 if self.sp.rtl else 0.032), 0.25, 2.2)
        self.ar = self.sp.rtl
        self.size = size

    def register(self):
        if not self.sfx_on:
            return
        if self.mode == 'write':
            sfx(self.t0, getattr(self, 'wsnd', None) or WRITE_SFX, self.dur, 0.9 if self.size >= 30 else 0.6)
        elif self.mode == 'pop':
            sfx(self.t0, 'pop', 0, 0.7)
        else:
            sfx(self.t0, 'soft', 0, 0.5)

    def draw(self, fr, t):
        a = self.fade(t)
        k = t - self.t0
        if self.mode == 'write':
            p = k / self.dur
            im = self.sp.revealed(p) if p < 1 else self.sp.im
            paste(fr, im, self.cx, self.cy, a, rot=self.rot)
        elif self.mode == 'pop':
            s = 0.4 + 0.6 * eback(k / 0.38)
            paste(fr, self.sp.im, self.cx, self.cy, a * clamp(k / 0.15), s, self.rot)
        else:
            paste(fr, self.sp.im, self.cx, self.cy + 14 * (1 - eo(k / 0.4)), a * eo(k / 0.4), rot=self.rot)

def P(cx, cy, s, pts):

def arcpts(cx, cy, rx, ry, a0, a1, n=40):
    return [(cx + rx * math.cos(math.radians(a0 + (a1 - a0) * i / n)),
             cy + ry * math.sin(math.radians(a0 + (a1 - a0) * i / n))) for i in range(n + 1)]

@functools.lru_cache(maxsize=512)
def icon_sprite(name, size=80, col=NAVY):
    S = 2
    im = Image.new('RGBA', (size * S * 2, size * S * 2), (0, 0, 0, 0))
    d = ImageDraw.Draw(im)
    draw_icon(d, name, size * S, size * S, size * S * 0.5, col, max(3, int(size / 16)) * S)
    im = im.resize((size * 2, size * 2), Image.LANCZOS)
    return im.crop(im.getbbox() or (0, 0, 2, 2))


def text_on(d, xy, text, fn, size, col, S=2):
    f = font(fn, size * S)
    kw = dict(direction='rtl', language='ar') if is_ar(text) and fn in ('ar', 'arb') else {}
    bb = d.textbbox((0, 0), text, font=f, **kw)
    x = xy[0] * S - (bb[0] + bb[2]) / 2
    y = xy[1] * S - (bb[1] + bb[3]) / 2
    d.text((x, y), text, font=f, fill=col, **kw)


@functools.lru_cache(maxsize=1024)
def card_sprite(w, h, icon, label, arabic, fill, border, lsize, icol, isize, label_col, ar_size):
    S = 2
    im = Image.new('RGBA', (w * S, h * S), (0, 0, 0, 0))
    d = ImageDraw.Draw(im)
    d.rounded_rectangle((2, 2, w * S - 3, h * S - 3), radius=12 * S, fill=fill + (255,), outline=border + (255,), width=3 * S)
    has_label = bool(label)
    nl = len(label.split('\n')) if label else 0
    lab_h = (nl * lsize * 1.08 + 12) if has_label else 0
    icy = (h - lab_h) / 2 + (3 if has_label else 0)
    if icon:
        isz = isize or int(min(w, h - lab_h) * 0.62)
        sc = isz * 0.72
        sc = min(sc, (h - lab_h) * 0.46, w * 0.42)
        draw_icon(d, icon, w * S / 2, icy * S, sc * S, icol, max(3, int(sc) // 9) * S)
    if arabic:
        text_on(d, (w / 2, icy + 4), arabic, 'ar', ar_size, NAVY + (255,))
    if has_label:
        lines = label.split('\n')
        n = len(lines)
        for i, ln in enumerate(lines):
            yy = h - 12 - lsize * 0.62 * (n - i) - (n - 1 - i) * lsize * 0.42 + lsize * 0.3
            text_on(d, (w / 2, yy), ln, 'kb', lsize, label_col + (255,))
    im = im.resize((w, h), Image.LANCZOS)
    return shadowed(im, (3, 4), 4, 55)

        a = self.fade(t)
        if self.anim == 'pop':
            s = 0.5 + 0.5 * eback(k / 0.4)

        else:
            paste(fr, self.sp, self.cx, self.cy - 30 * (1 - eo(k / 0.4)), a * clamp(k / 0.2), 1, self.rot)

def poly_len(pts):
    return sum(math.dist(pts[i], pts[i + 1]) for i in range(len(pts) - 1))


def poly_partial(pts, p):
    if p >= 1:
        return pts
    tot = poly_len(pts) * p
    out = [pts[0]]
    for i in range(len(pts) - 1):
        seg = math.dist(pts[i], pts[i + 1])
        if seg >= tot:
            f = tot / seg if seg else 0
            out.append((pts[i][0] + (pts[i + 1][0] - pts[i][0]) * f, pts[i][1] + (pts[i + 1][1] - pts[i][1]) * f))
            return out
        tot -= seg
        out.append(pts[i + 1])
    return out


class Line(El):
    z = 1

    def __init__(self, t0, pts, color=NAVY, width=4, dur=0.5, arrow=False, dashed=False, snd='swoosh', head=14, **kw):
        super().__init__(t0, **kw)
        self.pts, self.color, self.width, self.dur = pts, color, width, dur
        self.arrow, self.dashed, self.snd, self.head = arrow, dashed, snd, head

    def register(self):
        if self.sfx_on and self.snd:
            sfx(self.t0, self.snd, self.dur, 0.6)

    def draw(self, fr, t):
        p = eio((t - self.t0) / self.dur)
        a = self.fade(t)
        col = mix(PAPER, self.color, a)
        d = ImageDraw.Draw(fr)
        pts = poly_partial(self.pts, p)
        if len(pts) < 2:
            return
        if self.dashed:
            # draw dashes along
            L = poly_len(pts)
            n = int(L / 16)
            for i in range(n):
                q0, q1 = i / max(1, n), (i + 0.55) / max(1, n)
                a0 = poly_partial(pts, q0)[-1]
                a1 = poly_partial(pts, q1)[-1]
                d.line([a0, a1], fill=col, width=self.width)
        else:
            d.line(pts, fill=col, width=self.width, joint='curve')
            r = self.width / 2 - 0.5
            for q in (pts[0], pts[-1]):
                d.ellipse((q[0] - r, q[1] - r, q[0] + r, q[1] + r), fill=col)
        if self.arrow and p > 0.85:
            (x0, y0), (x1, y1) = pts[-2], pts[-1]
            ang = math.atan2(y1 - y0, x1 - x0)
            hs = self.head
            l = (x1 - hs * math.cos(ang - 0.45), y1 - hs * math.sin(ang - 0.45))
            r_ = (x1 - hs * math.cos(ang + 0.45), y1 - hs * math.sin(ang + 0.45))
            d.line([l, (x1, y1), r_], fill=col, width=self.width, joint='curve')


def arrow(t0, x0, y0, x1, y1, color=NAVY, bend=0.0, **kw):
    n = 24
    mx, my = (x0 + x1) / 2, (y0 + y1) / 2
    nx, ny = -(y1 - y0), (x1 - x0)
    L = math.hypot(nx, ny) or 1
    cxp, cyp = mx + nx / L * bend, my + ny / L * bend
    pts = []
    for i in range(n + 1):
        u = i / n
        pts.append(((1 - u) ** 2 * x0 + 2 * (1 - u) * u * cxp + u * u * x1, (1 - u) ** 2 * y0 + 2 * (1 - u) * u * cyp + u * u * y1))
    kw.setdefault('arrow', True)
    return Line(t0, pts, color, **kw)


class Ellipse(Line):
    def __init__(self, t0, cx, cy, rx, ry, color=GREEN, width=4, dur=0.5, **kw):
        pts = []
        n = 70
        for i in range(n + 1):
            a = math.radians(-120 + 385 * i / n)
            wob = 1 + 0.04 * math.sin(i / n * 7)
            pts.append((cx + rx * wob * math.cos(a), cy + ry * wob * math.sin(a)))
        kw.setdefault('snd', MARKER_SFX)
        super().__init__(t0, pts, color, width, dur, **kw)


def strike(t0, x0, x1, y, color=RED, **kw):
    pts = [(x0, y + 2), ((x0 + x1) / 2, y - 3), (x1, y + 1)]
    kw.setdefault('snd', 'scratch')
    kw.setdefault('dur', 0.3)
    return Line(t0, pts, color, 4, **kw)


def underline(t0, x0, x1, y, color=GREEN, **kw):
    pts = [(x0, y), ((x0 + x1) / 2, y + 3), (x1, y - 1)]
    kw.setdefault('snd', 'scratch')
    kw.setdefault('dur', 0.35)
    return Line(t0, pts, color, 4, **kw)


class Mark(El):
    """check or cross drawn progressively"""
    z = 2

    def __init__(self, t0, kind, cx, cy, s=26, **kw):
        super().__init__(t0, **kw)
        self.kind, self.cx, self.cy, self.s = kind, cx, cy, s

    def register(self):
        if self.sfx_on:
            sfx(self.t0, 'correct' if self.kind == 'check' else 'wrong', 0, 0.75)

    def draw(self, fr, t):
        k = (t - self.t0) / 0.3
        a = self.fade(t)
        d = ImageDraw.Draw(fr)
        cx, cy, s = self.cx, self.cy, self.s
        if self.kind == 'check':
            col = mix(PAPER, GREEN, a)
            pts = [(cx - s, cy), (cx - s * .3, cy + s * .7), (cx + s, cy - s * .8)]
            pp = poly_partial(pts, eo(k))
            if len(pp) > 1:
                d.line(pp, fill=col, width=int(s / 4.5), joint='curve')
        else:
            col = mix(PAPER, RED, a)
            p1 = poly_partial([(cx - s, cy - s), (cx + s, cy + s)], eo(k * 2))
            p2 = poly_partial([(cx + s, cy - s), (cx - s, cy + s)], eo(k * 2 - 1))
            for pp in (p1, p2):
                if len(pp) > 1:
                    d.line(pp, fill=col, width=int(s / 4.5))


class Ticks(El):
    def __init__(self, t0, n, total=15, cy=196, x0=95, x1=625, **kw):
        super().__init__(t0, **kw)
        self.n, self.total, self.cy, self.x0, self.x1 = n, total, cy, x0, x1
        self.pencil = pencil_sprite()

    def register(self):
        sfx(self.t0, 'soft', 0, 0.5)
        sfx(self.t0 + 0.35, 'tick', 0, 1.0)
        sfx(self.t0 + 0.35, 'write', 0.35, 0.7)

    def xs(self, i):
        return self.x0 + (self.x1 - self.x0) * i / (self.total - 1)

    def draw(self, fr, t):
        k = t - self.t0
        a = self.fade(t) * clamp(k / 0.25)
        d = ImageDraw.Draw(fr)
        for i in range(self.total):
            x = self.xs(i)
            if i < self.n - 1:
                col = mix(PAPER, NAVY, a)
                d.line([(x, self.cy - 22), (x, self.cy + 22)], fill=col, width=5)
            elif i == self.n - 1:
                p = eo((k - 0.35) / 0.35)
                if p > 0:
                    col = mix(PAPER, GREEN, a)
                    d.line([(x, self.cy - 22), (x, self.cy - 22 + 44 * p)], fill=col, width=6)
                if k < 1.3:
                    pa = a * clamp((1.3 - k) / 0.3) * clamp(k / 0.2)
                    paste(fr, self.pencil, x + 14, self.cy - 30 + 44 * clamp((k - 0.35) / 0.35), pa)
            else:
                col = mix(PAPER, LGRAY, a)
                d.line([(x, self.cy - 22), (x, self.cy + 22)], fill=col, width=3)


@functools.lru_cache(None)
def pencil_sprite():

    im = Image.new('RGBA', (40 * S, 80 * S), (0, 0, 0, 0))

    d.polygon([(6 * S, 4 * S), (26 * S, 4 * S), (26 * S, 56 * S), (16 * S, 74 * S), (6 * S, 56 * S)], fill=(245, 200, 70), outline=NAVY, width=2 * S)

class Bar(El):
    """progress bar animating from p0 to p1"""

    def __init__(self, t0, cx, cy, w, h, p0, p1, dur=1.0, col=GREEN, **kw):
        super().__init__(t0, **kw)
        self.cx, self.cy, self.w, self.h, self.p0, self.p1, self.dur, self.col = cx, cy, w, h, p0, p1, dur, col

    def register(self):
        sfx(self.t0, 'rise', self.dur, 0.6)

    def draw(self, fr, t):
        k = t - self.t0
        a = self.fade(t) * clamp(k / 0.2)
        d = ImageDraw.Draw(fr)
        x0, y0 = self.cx - self.w / 2, self.cy - self.h / 2
        p = self.p0 + (self.p1 - self.p0) * eio(k / self.dur)
        d.rounded_rectangle((x0, y0, x0 + self.w, y0 + self.h), radius=self.h // 2, fill=mix(PAPER, WHITE, a), outline=mix(PAPER, NAVY, a), width=3)
        if p > 0.02:
            d.rounded_rectangle((x0 + 4, y0 + 4, x0 + 4 + (self.w - 8) * p, y0 + self.h - 4), radius=(self.h - 8) // 2, fill=mix(PAPER, self.col, a))


class Ruler(El):
    def __init__(self, t0, cx, cy, w=560, h=24, **kw):
        super().__init__(t0, **kw)
        self.cx, self.cy, self.w, self.h = cx, cy, w, h

    def register(self):
        sfx(self.t0, 'slide', 0.4, 0.7)

    def draw(self, fr, t):
        k = t - self.t0
        p = eo(k / 0.45)
        a = self.fade(t)
        d = ImageDraw.Draw(fr)
        x0 = self.cx - self.w / 2
        y0 = self.cy - self.h / 2
        ww = self.w * p
        if ww < 4:
            return
        d.rectangle((x0, y0, x0 + ww, y0 + self.h), fill=mix(PAPER, (240, 208, 100), a), outline=mix(PAPER, (190, 150, 60), a), width=2)
        for i in range(int(ww / 10)):
            x = x0 + i * 10
            hh = self.h * (0.55 if i % 5 == 0 else 0.3)
            d.line([(x, y0), (x, y0 + hh)], fill=mix(PAPER, (150, 110, 40), a), width=1)


class Wave(Line):
    def __init__(self, t0, x0, x1, cy, amp, freq, color=NAVY, grow=False, **kw):
        pts = []
        n = 120
        for i in range(n + 1):
            u = i / n
            A = amp * (u * 1.6 if grow else 1)
            pts.append((x0 + (x1 - x0) * u, cy + A * math.sin(u * freq * 2 * math.pi)))
        kw.setdefault('snd', 'swoosh')
        super().__init__(t0, pts, color, 4, **kw)


class Box(El):
    """plain rounded rectangle frame (draw-on fade)"""
    z = -1

    def __init__(self, t0, x0, y0, x1, y1, col=NAVY, fill=None, width=3, r=14, dashed=False, **kw):
        super().__init__(t0, **kw)
        self.b = (x0, y0, x1, y1)
        self.col, self.fill, self.width, self.r = col, fill, width, r

    def register(self):
        if self.sfx_on:
            sfx(self.t0, 'soft', 0, 0.6)

    def draw(self, fr, t):
        k = t - self.t0
        a = self.fade(t) * eo(k / 0.3)
        s = 0.92 + 0.08 * eo(k / 0.3)
        x0, y0, x1, y1 = self.b
        cx, cy = (x0 + x1) / 2, (y0 + y1) / 2
        bx = (cx - (cx - x0) * s, cy - (cy - y0) * s, cx + (x1 - cx) * s, cy + (y1 - cy) * s)
        d = ImageDraw.Draw(fr)
        d.rounded_rectangle(bx, radius=self.r, fill=mix(PAPER, self.fill, a) if self.fill else None,
                            outline=mix(PAPER, self.col, a), width=self.width)


class Dots(El):
    def __init__(self, t0, pts, col=NAVY, r=6, gap=0.08, **kw):
        super().__init__(t0, **kw)
        self.pts, self.col, self.r, self.gap = pts, col, r, gap

    def register(self):
        if self.gap < 0.05:
            sfx(self.t0, 'rise', len(self.pts) * self.gap, 0.8)
            for i in range(0, len(self.pts), 6):
                sfx(self.t0 + i * self.gap, 'tick', 0, 0.3)
            return
        for i in range(len(self.pts)):
            sfx(self.t0 + i * self.gap, 'tick', 0, 0.35)

    def draw(self, fr, t):
        d = ImageDraw.Draw(fr)
        a = self.fade(t)
        for i, (x, y) in enumerate(self.pts):
            k = t - self.t0 - i * self.gap
            if k <= 0:
                continue
            r = self.r * eback(k / 0.25)
            col = mix(PAPER, self.col, a)
            d.ellipse((x - r, y - r, x + r, y + r), fill=col)

def _fx_mod(fx, k):
    """kembalikan (scale, rot, dy, alpha) untuk animasi ikon yang hidup"""
    if not fx or k < 0.45:
        return 1.0, 0.0, 0.0, 1.0
    u = k - 0.45
    if fx == 'beat':
        p = (u * 1.4) % 1.0
        return 1 + 0.09 * math.exp(-p * 7) * math.sin(p * 18), 0.0, 0.0, 1.0
    if fx == 'wiggle':
        return 1.0, 7 * math.sin(u * 2 * math.pi * 1.1) * math.exp(-u * 0.25), 0.0, 1.0
    if fx == 'float':
        return 1.0, 0.0, 6 * math.sin(u * 2 * math.pi * 0.6), 1.0
    if fx == 'twinkle':
        return 1 + 0.05 * math.sin(u * 9), 4 * math.sin(u * 3), 0.0, 0.8 + 0.2 * math.sin(u * 7)
    if fx == 'drip':
        return 1.0, 0.0, 4 * math.sin(u * 2 * math.pi * 0.8), 1.0
    return 1.0, 0.0, 0.0, 1.0


class Icon2(El):
    """ikon dengan animasi hidup: fx = beat | wiggle | float | twinkle | drip"""

    def __init__(self, t0, name, cx, cy, size=80, col=NAVY, rot=0, fx=None, snd='pop', **kw):
        super().__init__(t0, **kw)
        self.sp = icon_sprite(name, size, col)
        self.cx, self.cy, self.rot, self.fx, self.snd, self.name = cx, cy, rot, fx, snd, name

    def register(self):
        if self.sfx_on:
            sfx(self.t0, self.snd, 0, 0.7)

    def draw(self, fr, t):
        k = t - self.t0
        s0 = 0.3 + 0.7 * eback(k / 0.42)
        sm, rm, dy, am = _fx_mod(self.fx, k)
        paste(fr, self.sp, self.cx, self.cy + dy, self.fade(t) * clamp(k / 0.12) * am, s0 * sm, self.rot + rm)
        if self.fx == 'drip' and k > 0.5:
            u = ((k - 0.5) * 0.9) % 1.0
            d = ImageDraw.Draw(fr)
            y = self.cy + self.sp.height * 0.45 + 60 * eio(u)
            a = self.fade(t) * (1 - u)
            col = mix(PAPER, (90, 150, 220), a)
            d.ellipse((self.cx - 6, y - 9, self.cx + 6, y + 9), fill=col)


class Card2(Card):
    def __init__(self, *a, fx=None, **kw):
        super().__init__(*a, **kw)

                       (x, y + s * 1.6), (x - s * .45, y + s * .45), (x - s * 1.6, y), (x - s * .45, y - s * .45)], fill=col)


@functools.lru_cache(None)
def _sheet():
    im = Image.new('RGB', (W, H + 72))
    im.paste(BG, (0, 72)); im.paste(BG.crop((0, 0, W, 72)), (0, 0))
    sh = Image.new('RGBA', (60, H + 72), (0, 0, 0, 0))

class PageTurn(El):
    """transisi: lembar kertas baru menyapu layar (masuk dari kanan, keluar ke kiri)"""

    def draw(self, fr, t):

            fr.paste(sheet, (x, 0))

                fr.paste(shadow.transpose(Image.FLIP_LEFT_RIGHT), (x - 60, 0), shadow.transpose(Image.FLIP_LEFT_RIGHT))
            else:
                fr.paste(shadow, (x + W, 0), shadow)

def _poly_draw(fr, polys, p, col, width):
    """gambar sekumpulan polyline secara bertahap (p 0..1 dari total panjang)"""
    d = ImageDraw.Draw(fr)
    tot = sum(poly_len(pl) for pl in polys) or 1
    left = tot * p
    for pl in polys:
        L = poly_len(pl)
        if left <= 0:
            break
        part = pl if left >= L else poly_partial(pl, left / L)
        if len(part) > 1:
            d.line(part, fill=col, width=width, joint='curve')
        left -= L


def _arc(cx, cy, rx, ry, a0, a1, n=30):
    return [(cx + rx * math.cos(math.radians(a0 + (a1 - a0) * i / n)), cy + ry * math.sin(math.radians(a0 + (a1 - a0) * i / n))) for i in range(n + 1)]


def lineart(kind, cx, cy, s):
    """kumpulan polyline untuk ilustrasi garis (tanpa wajah)"""
    P_ = lambda pts: [(cx + x * s, cy + y * s) for x, y in pts]
    if kind == 'cave':
        return [_arc(cx, cy + .6 * s, 1.0 * s, 1.25 * s, 180, 360), _arc(cx, cy + .6 * s, .62 * s, .85 * s, 180, 360)]
    if kind == 'stick':
        return [_arc(cx, cy - .72 * s, .16 * s, .16 * s, 0, 360, 20), P_([(0, -.56), (0, .2)]), P_([(-.32, -.25), (0, -.35), (.32, -.25)]),
                P_([(-.25, .8), (0, .2), (.25, .8)])]
    if kind == 'buildings':
        return [P_([(-1, .8), (-1, -.1), (-.55, -.1), (-.55, .8)]), P_([(-.55, .8), (-.55, -.6), (-.05, -.6), (-.05, .8)]),
                P_([(-.05, .8), (-.05, -.25), (.45, -.25), (.45, .8)]), P_([(.45, .8), (.45, .1), (1, .1), (1, .8)]), P_([(-1.1, .8), (1.1, .8)])]
    if kind == 'house':
        return [P_([(-.7, .8), (-.7, -.05), (0, -.7), (.7, -.05), (.7, .8), (-.7, .8)]), P_([(-.18, .8), (-.18, .3), (.18, .3), (.18, .8)])]
    if kind == 'mosque':
        return [P_([(-.8, .8), (-.8, .05), (.55, .05), (.55, .8), (-.8, .8)]), _arc(-.12 * s + cx, cy + .05 * s, .45 * s, .55 * s, 180, 360),
                P_([(.7, .8), (.7, -.7), (.92, -.7), (.92, .8)]), P_([(.66, -.7), (.81, -.98), (.96, -.7)]),
                _arc(cx - .12 * s, cy + .62 * s, .18 * s, .22 * s, 180, 360) + P_([(.06, .8)])]
    if kind == 'mountains':
        return [P_([(-1.1, .6), (-.6, -.3), (-.3, .15), (0, -.55), (.55, .6)])]
    if kind == 'waves':
        return [[(cx + (-1 + 2 * i / 40) * s, cy + .12 * s * math.sin(i / 40 * 4 * math.pi)) for i in range(41)]]
    if kind == 'tree':
        return [P_([(0, .8), (0, 0)]), _arc(cx, cy - .25 * s, .3 * s, .3 * s, 0, 360, 24)]
    if kind == 'bird':
        return [_arc(cx - .2 * s, cy, .2 * s, .14 * s, 200, 340, 10) + _arc(cx + .2 * s, cy, .2 * s, .14 * s, 200, 340, 10)]
    if kind == 'sun':
        out = [_arc(cx, cy, .35 * s, .35 * s, 0, 360, 24)]
        for k in range(8):
            a = math.radians(k * 45)
            out.append([(cx + .5 * s * math.cos(a), cy + .5 * s * math.sin(a)), (cx + .7 * s * math.cos(a), cy + .7 * s * math.sin(a))])
        return out
    if kind == 'moon':
        return [_arc(cx, cy, .6 * s, .6 * s, 60, 300, 30) + _arc(cx + .25 * s, cy, .48 * s, .52 * s, 250, 110, 30)]
    if kind == 'paper':
        return [P_([(-.8, -.6), (.8, -.6), (.8, .6), (-.8, .6), (-.8, -.6)])] + [P_([(-.55, y), (.55, y)]) for y in (-.3, -.05, .2)]
    if kind == 'ground':
        return [P_([(-1, 0), (1, 0)])]
    raise KeyError(kind)

class Walker(El):
    """orang lidi berjalan di sepanjang lintasan, meninggalkan jejak titik"""
    z = 2

    def __init__(self, t0, path, dur=2.5, s=34, color=NAVY, trail=True, **kw):

        self.path, self.dur, self.s, self.color, self.trail = path, dur, s, color, trail
        self.L = poly_len(path)

    def register(self):
        n = int(self.dur / 0.35)
        for i in range(n):
            sfx(self.t0 + i * 0.35, 'tick', 0, 0.25)

    def draw(self, fr, t):
        k = clamp((t - self.t0) / self.dur)

        if self.trail:
            n = int(self.L / 26)
            for i in range(n):
                u = i / max(1, n)
                if u > k - 0.02:
                    break
                x, y = poly_partial(self.path, u)[-1]
                d.ellipse((x - 3.5, y - 3.5, x + 3.5, y + 3.5), fill=mix(PAPER, GRAY, a))
        x, y = poly_partial(self.path, max(0.001, k))[-1]
        bob = 3 * abs(math.sin(k * self.dur * 6)) if k < 1 else 0
        polys = lineart('stick', x, y - self.s * .8 - bob, self.s)
        if k < 1:  # kaki melangkah
            sw = .25 * math.sin(k * self.dur * 12)
            polys[3] = [(x - self.s * sw - self.s * .2, y - bob), (x, y - self.s * .6 - bob), (x + self.s * sw + self.s * .2, y - bob)]
        _poly_draw(fr, polys, 1, mix(PAPER, self.color, a), 4)

class CompareBars(El):
    """batang perbandingan bertumpuk dengan ikon, label, dan kurung selisih"""

    def __init__(self, t0, x0, x1, y, rows, gap=70, diff=None, dur=0.9, stagger=0.6, **kw):
        super().__init__(t0, **kw)
        self.x0, self.x1, self.y, self.rows, self.gap, self.diff, self.dur, self.stagger = x0, x1, y, rows, gap, diff, dur, stagger

    def register(self):
        for i in range(len(self.rows)):
            sfx(self.t0 + i * self.stagger, 'rise', self.dur, 0.5)
        if self.diff:
            sfx(self.t0 + len(self.rows) * self.stagger + .3, 'pop', 0, 0.6)

    def draw(self, fr, t):
        a = self.fade(t)
        d = ImageDraw.Draw(fr)
        L = self.x1 - self.x0 - 60
        ends = []
        for i, (icon, label, frac, col) in enumerate(self.rows):
            k = (t - self.t0 - i * self.stagger) / self.dur
            if k <= 0:
                continue
            y = self.y + i * self.gap
            paste(fr, icon_sprite(icon, 40, NAVY), self.x0 + 20, y, a * clamp(k * 3))
            w = L * frac * eio(k)
            if w > 2:
                d.rounded_rectangle((self.x0 + 50, y - 7, self.x0 + 50 + w, y + 7), radius=7, fill=mix(PAPER, col, a))
            if k > .7:
                paste(fr, tsprite(label, 'kr', 18, NAVY).im, self.x0 + 50 + L * frac / 2, y + 24, a * clamp((k - .7) * 4))
            ends.append((self.x0 + 50 + L * frac, y))
        if self.diff and len(ends) >= 2:
            k = (t - self.t0 - len(self.rows) * self.stagger - .2) / .4
            if k > 0:
                (xa, ya), (xb, yb) = ends[0], ends[-1]
                xr = max(xa, xb) + 14
                pts = poly_partial([(xa + 4, ya), (xr, ya), (xr, yb), (xb + 4, yb)], eo(k))
                if len(pts) > 1:
                    d.line(pts, fill=mix(PAPER, RED, a), width=3)
                if k > .8:
                    paste(fr, tsprite(self.diff, 'kb', 24, RED).im, xr + 10 + font('kb', 24).getlength(self.diff) / 2, (ya + yb) / 2, a)

COINC, COIND = (240, 178, 52), (176, 120, 30)
JAR, JARD = (32, 96, 88), (20, 64, 58)
SKY = (120, 160, 220)


def _coin(d, cx, cy, r, a=1.0, flat=False):
    if flat:  # koin dilihat dari samping (tumpukan)
        d.rounded_rectangle((cx - r, cy - r * .32, cx + r, cy + r * .32), radius=int(r * .3), fill=mix(PAPER, COINC, a), outline=mix(PAPER, COIND, a), width=2)
        return
    d.ellipse((cx - r, cy - r, cx + r, cy + r), fill=mix(PAPER, COINC, a), outline=mix(PAPER, COIND, a), width=max(2, int(r * .14)))
    ri = r * .58
    d.ellipse((cx - ri, cy - ri, cx + ri, cy + ri), outline=mix(PAPER, COIND, a * .8), width=max(1, int(r * .09)))


def _dots_arc(d, cx, cy, r, a0, a1, col, step=7, rad=2):
    for ang in range(int(a0), int(a1) + 1, step):
        x = cx + r * math.cos(math.radians(ang)); y = cy + r * math.sin(math.radians(ang))
        d.ellipse((x - rad, y - rad, x + rad, y + rad), fill=col)


class CoinBars(El):
    """tumpukan koin bernomor yang tumbuh dari kecil ke besar + panah tren"""

    def __init__(self, t0, x0, x1, ybase, n=7, maxc=11, dur=1.6, arrow=True, nums=True, **kw):
        super().__init__(t0, **kw)
        self.x0, self.x1, self.yb, self.n, self.maxc, self.dur, self.arrow, self.nums = x0, x1, ybase, n, maxc, dur, arrow, nums

    def register(self):
        for i in range(self.n):
            sfx(self.t0 + i * self.dur / self.n, 'coin', 0, 0.55)
        if self.arrow:
            sfx(self.t0 + self.dur + .1, 'swoosh', 0.4, 0.5)

    def draw(self, fr, t):
        a = self.fade(t); d = ImageDraw.Draw(fr)
        k0 = t - self.t0
        gap = (self.x1 - self.x0) / self.n; r = gap * .36
        d.line([(self.x0 - 10, self.yb + r * .4), (self.x1 + 10, self.yb + r * .4)], fill=mix(PAPER, NAVY, a * clamp(k0 / .3)), width=3)
        tops = []
        for i in range(self.n):
            cx = self.x0 + gap * (i + .5)
            target = max(1, round(self.maxc * (i + 1) / self.n))
            k = clamp((k0 - i * self.dur / self.n * .7) / (self.dur * .55))
            c = int(round(target * eo(k)))
            for j in range(c):
                _coin(d, cx, self.yb - j * r * .62, r, a, flat=True)
            tops.append((cx, self.yb - max(c - 1, 0) * r * .62 - r * .9))
            if self.nums and k > 0:
                yy = self.yb + r * .4 + 22
                d.ellipse((cx - 11, yy - 11, cx + 11, yy + 11), fill=mix(PAPER, JAR, a * clamp(k * 4)))
                paste(fr, tsprite(str(i + 1), 'kb', 15, WHITE).im, cx, yy, a * clamp(k * 4))
        if self.arrow:
            k = clamp((k0 - self.dur - .1) / .6)
            if k > 0:
                pts = [(x, y - 26) for x, y in tops]
                pp = poly_partial(pts, eio(k))
                if len(pp) > 1:
                    d.line(pp, fill=mix(PAPER, GREEN, a), width=4, joint='curve')
                if k >= 1:
                    (xa, ya), (xb, yb) = pts[-2], pts[-1]
                    ang = math.atan2(yb - ya, xb - xa)
                    for s_ in (2.5, -2.5):
                        d.line([(xb, yb), (xb - 16 * math.cos(ang + s_ / 6), yb - 16 * math.sin(ang + s_ / 6))], fill=mix(PAPER, GREEN, a), width=4)


class CoinJar(El):
    """toples: koin melayang masuk dari kiri-kanan lalu bertumpuk"""

    def __init__(self, t0, cx, cy, s=110, n=6, gap=0.45, fill=None, **kw):
        super().__init__(t0, **kw)
        self.cx, self.cy, self.s, self.n, self.gap = cx, cy, s, n, gap
        self.fillcol = fill or JAR

    def register(self):
        sfx(self.t0, 'pop', 0, 0.5)
        for i in range(self.n):
            sfx(self.t0 + .5 + i * self.gap + .35, 'coin', 0, 0.7)

    def slots(self):
        s = self.s; r = s * .2; out = []; row = 0; per = 3
        while len(out) < self.n:
            m = max(1, per - row)
            for j in range(m):
                out.append((self.cx + (j - (m - 1) / 2) * r * 2.05, self.cy + s * .62 - r - row * r * 1.75))
            row += 1
        return out[:self.n], r

    def draw(self, fr, t):
        a = self.fade(t) * clamp((t - self.t0) / .3); d = ImageDraw.Draw(fr)
        s = self.s; x0, x1, y0, y1 = self.cx - s * .62, self.cx + s * .62, self.cy - s * .55, self.cy + s * .7
        d.rounded_rectangle((x0, y0, x1, y1), radius=int(s * .16), fill=mix(PAPER, self.fillcol, a), outline=mix(PAPER, JARD, a), width=3)
        d.rounded_rectangle((x0 - 10, y0 - 14, x1 + 10, y0 + 2), radius=8, fill=mix(PAPER, (140, 200, 180), a))
        sl, r = self.slots()
        for i, (tx, ty) in enumerate(sl):
            k = (t - self.t0 - .5 - i * self.gap) / .4
            if k <= 0:
                continue
            side = -1 if i % 2 == 0 else 1
            sx, sy = self.cx + side * s * 1.3, self.cy - s * .3
            if k < 1:
                e = eio(k); x = sx + (tx - sx) * e; y = sy + (ty - sy) * e - math.sin(e * math.pi) * s * .8
            else:
                x, y = tx, ty
            _coin(d, x, y, r, a)


class RainGrow(El):
    """awan menurunkan hujan, tunas tumbuh lalu berbunga koin"""

    def __init__(self, t0, cx, ybase, s=120, dur=3.0, bloom=True, **kw):
        super().__init__(t0, **kw)
        self.cx, self.yb, self.s, self.dur, self.bloom = cx, ybase, s, dur, bloom

    def register(self):
        sfx(self.t0, 'soft', 0, 0.6)
        for i in range(10):
            sfx(self.t0 + .5 + i * self.dur * .08, 'drip', 0, 0.35)
        if self.bloom:
            sfx(self.t0 + self.dur, 'ding', 0, 0.6)

    def draw(self, fr, t):
        a = self.fade(t); d = ImageDraw.Draw(fr); s = self.s; k0 = t - self.t0
        ca = a * clamp(k0 / .4); cy = self.yb - s * 2.1
        for (dx, dy, rr) in ((-.38, .05, .3), (0, -.12, .42), (.38, .05, .3), (0, .1, .3)):
            x, y, r = self.cx + dx * s, cy + dy * s, rr * s
            d.ellipse((x - r, y - r, x + r, y + r), fill=mix(PAPER, SKY, ca))
        d.line([(self.cx - s * .8, self.yb), (self.cx + s * .8, self.yb)], fill=mix(PAPER, NAVY, ca), width=3)
        if k0 > .4:
            for i in range(9):
                ph = ((k0 - .4) * 1.4 + i * .37) % 1.0
                x = self.cx + (i - 4) * s * .13 + math.sin(i * 7) * s * .05
                y = cy + s * .35 + ph * (self.yb - cy - s * .45)
                d.line([(x, y), (x - 1, y + 9)], fill=mix(PAPER, SKY, a * (1 - ph * .4)), width=3)
        g = clamp((k0 - .7) / (self.dur - .7))
        if g > 0:
            top = self.yb - s * 1.1 * eo(g)
            d.line([(self.cx, self.yb), (self.cx, top)], fill=mix(PAPER, GREEN, a), width=5)
            for j, (fy, sd) in enumerate(((.35, -1), (.6, 1))):
                kk = clamp((g - .3 - j * .2) * 4)
                if kk > 0:
                    y = self.yb - s * 1.1 * fy; L = s * .32 * eback(kk)
                    d.polygon([(self.cx, y), (self.cx + sd * L, y - L * .45), (self.cx + sd * L * .5, y + L * .1)], fill=mix(PAPER, (120, 190, 130), a))
            if self.bloom and g >= 1:
                kb = clamp((k0 - self.dur) / .4)
                _coin(d, self.cx, top - 4, s * .17 * eback(kb), a)

        pts = [(self.cx + self.r * math.cos(math.radians(-90 + i * 360 / self.n)), self.cy + self.r * math.sin(math.radians(-90 + i * 360 / self.n))) for i in range(self.n)]
        kr = clamp((k0 - .3 - self.n * self.gap - .1) / .6)
        if kr > 0:
            ring = pts + [pts[0]]
            pp = poly_partial(ring, eio(kr))
            if len(pp) > 1:
                d.line(pp, fill=mix(PAPER, GRAY, a), width=2)
        for i, (x, y) in enumerate(pts):
            k = clamp((k0 - .3 - i * self.gap) / .35)
            if k <= 0:
                continue

            d.line([(self.cx, self.cy), (ex, ey)], fill=mix(PAPER, NAVY, a * .7), width=3)
            if k >= 1:
                kb = clamp((k0 - .3 - i * self.gap - .35) / .25)
                rr = 13 * eback(kb)
                d.ellipse((x - rr, y - rr, x + rr, y + rr), fill=mix(PAPER, cols[i % 3], a))
                if self.labs and i < len(self.labs):
                    ox = 1 if x > self.cx + 5 else (-1 if x < self.cx - 5 else 0)
                    oy = 1 if y > self.cy + 5 else -1
                    lw = self.labs[i].w
                    paste(fr, self.labs[i].im, x + ox * (lw / 2 + 22), y + (oy * 30 if ox == 0 else 0), a * kb)
        kc = clamp(k0 / .35)
        if self.center:
            paste(fr, icon_sprite(self.center, 70, NAVY), self.cx, self.cy, a * kc, .6 + .4 * eback(kc))
        else:
            _coin(d, self.cx, self.cy, 30 * eback(kc), a)

@functools.lru_cache(maxsize=2048)
def _wordim(w_, fn, size, col):
    """kata dengan tinggi baris tetap agar garis dasar sejajar"""
    S = 2; f = font(fn, size * S); asc, desc = f.getmetrics()
    im = Image.new('RGBA', (int(f.getlength(w_)) + 8 * S, asc + desc + 4 * S), (0, 0, 0, 0))
    ImageDraw.Draw(im).text((4 * S, 2 * S), w_, font=f, fill=col + (255,))
    return im.resize((im.width // S, im.height // S), Image.LANCZOS)


class Reveal(El):
    """kalimat muncul kata demi kata (abu-abu -> tinta), kata kunci berwarna"""

    def __init__(self, t0, text, cx, cy, size=30, maxw=600, per=0.16, hl=(), hlcol=GREEN, color=NAVY, fn='kb', times=None, **kw):
        super().__init__(t0, **kw)
        self.cx, self.cy, self.size, self.per, self.color, self.hlcol, self.fn = cx, cy, size, per, color, hlcol, fn
        words = text.split()
        hls = {h.lower() for h in hl}
        f = font(fn, size); sp = f.getlength(' ')
        lines, cur, cw = [], [], 0
        for w_ in words:
            ww = f.getlength(w_)
            if cur and cw + sp + ww > maxw:
                lines.append(cur); cur, cw = [], 0
            cur.append(w_); cw += (sp if cw else 0) + ww
        if cur:
            lines.append(cur)
        self.items = []
        lh = size * 1.35; y0 = cy - (len(lines) - 1) * lh / 2; idx = 0
        for li, ln in enumerate(lines):
            tw = sum(f.getlength(w_) for w_ in ln) + sp * (len(ln) - 1); x = cx - tw / 2
            for w_ in ln:
                ww = f.getlength(w_)
                key = w_.strip('.,!?;:"\'').lower()
                hit = key in hls
                tt = times[idx] if times else t0 + idx * per
                self.items.append((tt, x + ww / 2, y0 + li * lh, _wordim(w_, fn, size, hlcol if hit else color),
                                   _wordim(w_, fn, size, (190, 192, 200)), hit))
                x += ww + sp; idx += 1

    def register(self):
        sfx(self.t0, 'soft', 0, 0.4)

    def draw(self, fr, t):
        a = self.fade(t)
        for tt, x, y, im, gim, hit in self.items:
            k = (t - tt) / .25
            if k <= -0.6:
                continue
            if k < 0:
                paste(fr, gim, x, y + 3, a * (k + .6) / .6 * .7)
            else:
                paste(fr, im, x, y + 3 * (1 - eo(k)), a * clamp(k), 1.0 + (.12 * (1 - eo(k)) if hit else 0))

def _card_im(text, w=58, h=64, fn='kb', size=26, col=NAVY, fill=WHITE, border=NAVY):
    S = 2; im = Image.new('RGBA', (w * S, h * S), (0, 0, 0, 0)); d = ImageDraw.Draw(im)
    d.rounded_rectangle((2, 2, w * S - 3, h * S - 3), radius=6 * S, fill=fill + (255,), outline=border + (255,), width=3 * S)
    im = im.resize((w, h), Image.LANCZOS)
    sp = tsprite(text, fn, size, col)
    im.paste(sp.im, (int(w / 2 - sp.im.width / 2), int(h / 2 - sp.im.height / 2)), sp.im)
    return im

class _Mixed:
    def __init__(self, text, size, color):
        parts = text.split('ﷺ'); a = tsprite(parts[0].rstrip(), 'kb', size, color); b = tsprite('ﷺ', 'ar', int(size * 1.2), color)
        self.w, self.h = int(a.w + 4 + b.w), int(max(a.h, b.h))

        self.im = im


def label_sprite(text, size=24, color=NAVY, fn='kb'):

            d.rectangle((x0, y0, x1, y1), outline=mix(PAPER, self.color, a), width=3, fill=mix(PAPER, LGREEN, a * kf))
            paste(fr, self.sp.im, self.cx, self.cy, a, .7 + .3 * eback(kf))
            kr = clamp((t - self.tf - .3) / .4)
            if kr > 0:
                for ang in (-30, 0, 30):
                    r0, r1 = 14, 14 + 22 * eo(kr)
                    ca, sa = math.cos(math.radians(ang)), math.sin(math.radians(ang))
                    d.line([(x1 + r0 * ca, self.cy + r0 * sa), (x1 + r1 * ca, self.cy + r1 * sa)], fill=mix(PAPER, self.color, a), width=3)


class CrossOut(El):
    """coretan silang merah tipis di atas kartu/kotak (pilihan yang bukan jawabannya)"""
    z = 3

    def __init__(self, t0, x0, y0, x1, y1, color=RED, **kw):
        super().__init__(t0, **kw)
        self.b, self.color = (x0, y0, x1, y1), color

    def register(self):
        sfx(self.t0, 'scratch', 0, 0.5)

    def draw(self, fr, t):
        a = self.fade(t); x0, y0, x1, y1 = self.b
        _poly_draw(fr, [[(x0, y1), (x1, y0)]], eo((t - self.t0) / .3), mix(PAPER, self.color, a), 3)


class Crowd(El):
    """titik-titik orang muncul berkelompok (umat yang bertanya / hamba-hamba)"""

    def __init__(self, t0, cx, cy, n=8, w=180, gap=0.08, **kw):
        super().__init__(t0, **kw)
        self.cx, self.cy, self.n, self.w, self.gap = cx, cy, n, w, gap

    def register(self):
        for i in range(0, self.n, 2):
            sfx(self.t0 + i * self.gap, 'tick', 0, 0.3)

    def draw(self, fr, t):
        a = self.fade(t); d = ImageDraw.Draw(fr)
        for i in range(self.n):
            k = clamp((t - self.t0 - i * self.gap) / .2)
            if k <= 0:
                continue
            x = self.cx - self.w / 2 + self.w * i / max(1, self.n - 1)
            y = self.cy + (math.sin(i * 2.3) * 8)
            r = 4.5 * eback(k)
            d.ellipse((x - r, y - r, x + r, y + r), fill=mix(PAPER, NAVY, a))
