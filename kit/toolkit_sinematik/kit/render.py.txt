import sys, subprocess, os, importlib
import engine
from engine import BG, W, H, FPS
import subs

script = importlib.import_module(os.environ.get('SCRIPT', 'script'))
ELS = sorted(script.ELS, key=lambda e: (e.z, e.t0))
TOTAL = script.TOTAL


from PIL import Image, ImageDraw
SHIFT = 72
BG2 = Image.new('RGB', (W, H + SHIFT))
BG2.paste(BG, (0, SHIFT))
BG2.paste(BG.crop((0, 0, W, SHIFT)), (0, 0))


import numpy as np
CINE = os.environ.get('CINE', '0')  # 0 = polos; 1 = hangat lembut; 2 = film; 3 = dramatis
_VIG = {}


def _vig(strength):
    if strength not in _VIG:
        yy, xx = np.mgrid[0:H, 0:W]
        d = np.sqrt(((xx - W / 2) / (W * 0.62)) ** 2 + ((yy - H / 2) / (H * 0.62)) ** 2)
        _VIG[strength] = (1 - strength * np.clip(d - 0.35, 0, 1) ** 1.6)[..., None]
    return _VIG[strength]


def cinematic(fr, t):
    if CINE == '0':
        return fr
    a = np.asarray(fr).astype(np.float32)
    lvl = int(CINE)
    # grading hangat: sedikit merah/kuning di highlight, sedikit hijau-biru di bayangan
    a = a * np.array([1.03, 1.0, 0.95], np.float32) + np.array([4, 2, -2], np.float32)
    a = a * _vig({1: 0.35, 2: 0.45, 3: 0.6}[lvl])
    if lvl >= 2:
        g = Image.fromarray(np.clip(a, 0, 255).astype(np.uint8)).resize((W // 4, H // 4)).resize((W, H), Image.BILINEAR)
        a = a * 0.86 + np.asarray(g).astype(np.float32) * 0.14 + 6
        rng = np.random.default_rng(int(t * 30))
        a += rng.normal(0, 5.5, (H // 2, W // 2, 1)).repeat(2, 0).repeat(2, 1)
        leak = np.linspace(1, 0, W, dtype=np.float32)[None, :, None] * np.linspace(1, 0.2, H, dtype=np.float32)[:, None, None]
        a += leak * np.array([18, 10, 0], np.float32) * (0.6 + 0.4 * np.sin(t * 0.7))
    a = np.clip(a, 0, 255).astype(np.uint8)
    out = Image.fromarray(a)
    if lvl == 3:
        d = ImageDraw.Draw(out)
        d.rectangle((0, 0, W, 46), fill=(18, 20, 26)); d.rectangle((0, H - 46, W, H), fill=(18, 20, 26))
    return out


def frame(t):
    big = BG2.copy()
    for e in ELS:
        if e.active(t):
            e.draw(big, t)
    fr = big.crop((0, SHIFT, W, H + SHIFT))
    cam = getattr(script, 'CAM', None)
    if cam:
        fr = camera(fr, t, cam)
    if CINE == '3':
        z = 1.03 + 0.015 * np.sin(t * 0.35)
        w2, h2 = W / z, H / z
        fr = fr.transform((W, H), Image.EXTENT, ((W - w2) / 2, (H - h2) / 2, (W + w2) / 2, (H + h2) / 2), Image.BICUBIC)
    if os.environ.get('NOSUB', '1') != '1':  # bawaan TANPA subtitle (pilihan pengguna sejak episode 05); NOSUB=0 untuk menampilkan
        subs.draw_sub(fr, t)
    return cinematic(fr, t)


def cam_at(t, cam):
    from engine import eio
    if t <= cam[0][0]:
        return cam[0][1:]
    for (t0, z0, x0, y0), (t1, z1, x1, y1) in zip(cam, cam[1:]):
        if t0 <= t <= t1:
            u = eio((t - t0) / max(1e-3, t1 - t0))
            return (z0 + (z1 - z0) * u, x0 + (x1 - x0) * u, y0 + (y1 - y0) * u)
    return cam[-1][1:]


def camera(fr, t, cam):
    z, fx, fy = cam_at(t, cam)
    if z <= 1.001:
        return fr
    fy -= SHIFT
    w, h = W / z, H / z
    x0 = min(max(fx - w / 2, 0), W - w)
    y0 = min(max(fy - h / 2, 0), H - h)
    return fr.transform((W, H), Image.EXTENT, (x0, y0, x0 + w, y0 + h), Image.BICUBIC)


def render_range(f0, f1, out):
    p = subprocess.Popen(['ffmpeg', '-v', 'error', '-y', '-f', 'rawvideo', '-pix_fmt', 'rgb24', '-s', f'{W}x{H}',
                          '-r', str(FPS), '-i', '-', '-vf', 'scale=1080:1920:flags=lanczos+accurate_rnd+full_chroma_int,unsharp=5:5:0.6:5:5:0.0',
                          '-c:v', 'libx264', '-preset', 'medium', '-crf', '16', '-profile:v', 'high', '-level', '4.2',
                          '-maxrate', '14M', '-bufsize', '28M', '-g', '60', '-pix_fmt', 'yuv420p',
                          '-color_primaries', 'bt709', '-color_trc', 'bt709', '-colorspace', 'bt709', out], stdin=subprocess.PIPE)
    for f in range(f0, f1):
        p.stdin.write(frame(f / FPS).tobytes())
        if f % 900 == 0:
            print(out, f, flush=True)
    p.stdin.close()
    p.wait()


if __name__ == '__main__':
    if sys.argv[1] == 'png':
        os.makedirs('prev', exist_ok=True)
        for a in sys.argv[2:]:
            frame(float(a)).save(f'prev/p_{float(a):07.2f}.png')
    else:
        f0, f1, out = int(sys.argv[2]), int(sys.argv[3]), sys.argv[4]
        render_range(f0, f1, out)
