FW, FH = W, H + 72          # kanvas besar saat render (render.py menggeser 72 px)
EMER = (14, 61, 46); GOLDL = (222, 186, 98); CREAM = (250, 244, 226); WOOD = (150, 108, 72)


def _paper(w, h, col=(240, 239, 231)):
    im = Image.new('RGB', (w, h), col)
    d = ImageDraw.Draw(im)
    for x in range(0, w, 36):
        d.line([(x, 0), (x, h)], fill=(226, 225, 215))
    for y in range(0, h, 36):
        d.line([(0, y), (w, y)], fill=(226, 225, 215))
    return im


class FrameZoom(El):
    """seluruh layar mengecil menjadi lukisan berbingkai yang tergantung di dinding, lalu kembali membesar"""
    z = 60

    def __init__(self, t0, t1, scale=0.5, wall=(236, 226, 206), **kw):
        super().__init__(t0, out=t1, **kw)
        self.t1, self.scale, self.wall = t1, scale, wall

    def register(self):
        if self.sfx_on:
            sfx(self.t0, 'whoosh', 0, 0.5); sfx(self.t1 - 0.6, 'whoosh', 0, 0.4)

    def draw(self, fr, t):
        a = eio(clamp((t - self.t0) / 0.9)) * (1 - eio(clamp((t - (self.t1 - 0.8)) / 0.8)))
        if a <= 0.001:
            return
        s = 1 - (1 - self.scale) * a
        src = fr.copy()
        bg = Image.new('RGB', fr.size, self.wall)
        d = ImageDraw.Draw(bg)
        for y in range(0, fr.height, 90):                      # papan dinding
            d.line([(0, y), (fr.width, y)], fill=(222, 210, 186), width=2)
        d.rectangle((0, int(fr.height * 0.80), fr.width, fr.height), fill=(205, 192, 160))   # lantai tatami
        for x in range(0, fr.width, 120):
            d.line([(x, int(fr.height * 0.80)), (x, fr.height)], fill=(186, 172, 140), width=2)
        w, h = int(fr.width * s), int(fr.height * s)
        cx, cy = fr.width / 2, fr.height * (0.5 - 0.08 * a)
        sm = src.resize((w, h), Image.LANCZOS)
        b = int(26 * a)
        x0, y0 = int(cx - w / 2), int(cy - h / 2)
        d.rectangle((x0 - b - 6, y0 - b - 6, x0 + w + b + 6, y0 + h + b + 6), fill=WOOD)              # bingkai kayu
        d.rectangle((x0 - b, y0 - b, x0 + w + b, y0 + h + b), fill=EMER)                              # pasepartu zamrud
        d.line([(cx, y0 - b - 6), (cx - 60 * a, y0 - b - 70 * a), (cx + 60 * a, y0 - b - 70 * a), (cx, y0 - b - 6)], fill=WOOD, width=3)
        bg.paste(sm, (x0, y0))
        out = Image.blend(src, bg, min(1, a * 1.6)) if a < 0.6 else bg
        fr.paste(out)


class GridMultiply(El):
    """layar (atau isinya) mengecil lalu berlipat menjadi mozaik kartu; gray=True membuat semua abu-abu kecuali satu"""
    z = 60

    def __init__(self, t0, t1, n=6, gray=False, keep=(2, 3), **kw):
        super().__init__(t0, out=t1, **kw)
        self.t1, self.n, self.gray, self.keep = t1, n, gray, keep

    def register(self):
        if self.sfx_on:
            sfx(self.t0, 'swoosh', 0.6, 0.4)
            for i in range(6):
                sfx(self.t0 + 0.5 + i * 0.07, 'tick', 0, 0.2)

    def draw(self, fr, t):
        k = eio(clamp((t - self.t0) / 1.4))
        if k <= 0.001:
            return
        src = fr.copy()
        n = 1 + (self.n - 1) * k
        cw, ch = fr.width / n, fr.height / n
        card = src.resize((max(2, int(cw - 8)), max(2, int(ch - 8))), Image.LANCZOS)
        gcard = ImageOps.grayscale(card).convert('RGB') if self.gray else card
        bg = Image.new('RGB', fr.size, (196, 180, 140))
        cols = int(math.ceil(n)) + 2
        ox = fr.width / 2 - cw / 2 - cw * (cols // 2)
        oy = fr.height / 2 - ch / 2 - ch * (cols // 2)
        zk = clamp((t - self.t0 - 1.8) / 1.2) if self.gray else 0
        for i in range(cols):
            for j in range(cols):
                x, y = ox + i * cw, oy + j * ch
                if x > fr.width or y > fr.height or x + cw < 0 or y + ch < 0:
                    continue
                center = (i == cols // 2 and j == cols // 2)
                im = card if (center or not self.gray) else gcard
                bg.paste(im, (int(x + 4), int(y + 4)))
                if center and self.gray and k >= 1:
                    dd = ImageDraw.Draw(bg)
                    dd.rectangle((x + 1, y + 1, x + cw - 1, y + ch - 1), outline=GOLD, width=4)
        if zk > 0:   # kamera mendekat ke kartu berwarna di tengah
            z = 1 + (n - 1) * eio(zk)
            w2, h2 = fr.width / z, fr.height / z
            bg = bg.transform(fr.size, Image.EXTENT, ((fr.width - w2) / 2, (fr.height - h2) / 2, (fr.width + w2) / 2, (fr.height + h2) / 2), Image.BICUBIC)
        fr.paste(bg)


class SealStack(El):
    """stempel segel persegi berangka jatuh berurutan dari atas, lalu berganti warna (seperti hanko)"""

    COLS = [(30, 60, 110), (40, 120, 110), (100, 70, 130), (200, 90, 70), (210, 160, 60), (44, 130, 76), (150, 80, 60)]

    def __init__(self, t0, cx, y0, labels, gap=0.45, size=64, step=86, trecolor=None, ar=True, **kw):
        super().__init__(t0, **kw)
        self.cx, self.y0, self.labels, self.gap, self.size, self.step, self.tr, self.ar = cx, y0, labels, gap, size, step, trecolor, ar

    def register(self):
        if self.sfx_on:
            for i in range(len(self.labels)):
                sfx(self.t0 + i * self.gap + 0.25, 'thud', 0, 0.35)
            if self.tr:
                sfx(self.tr, 'ding', 0, 0.35)

    def _seal(self, lab, col):
        s = self.size
        im = Image.new('RGBA', (s + 8, s + 8), (0, 0, 0, 0))
        d = ImageDraw.Draw(im)
        rng = np.random.default_rng(len(lab) + s)
        pts = [(4 + rng.uniform(-2, 2), 4 + rng.uniform(-2, 2)), (s + 4 + rng.uniform(-2, 2), 4 + rng.uniform(-2, 2)),
               (s + 4 + rng.uniform(-2, 2), s + 4 + rng.uniform(-2, 2)), (4 + rng.uniform(-2, 2), s + 4 + rng.uniform(-2, 2))]
        d.polygon(pts, fill=col + (235,))
        sp = tsprite(lab, 'ar' if self.ar else 'kb', int(s * (0.62 if self.ar else 0.55)), CREAM).im
        im.paste(sp, (int((s + 8 - sp.width) / 2), int((s + 8 - sp.height) / 2) - (4 if self.ar else 0)), sp)
        return im.filter(ImageFilter.GaussianBlur(0.4))

    def draw(self, fr, t):
        a = self.fade(t)
        for i, lab in enumerate(self.labels):
            k = clamp((t - self.t0 - i * self.gap) / 0.35)
            if k <= 0:
                continue
            col = (196, 64, 52) if (self.tr is None or t < self.tr + i * 0.08) else self.COLS[i % len(self.COLS)]
            im = self._seal(lab, col)
            sc = 1.6 - 0.6 * eo(k)
            im = im.resize((int(im.width * sc), int(im.height * sc)), Image.BICUBIC)
            if a * k < 1:
                im.putalpha(im.getchannel('A').point(lambda v, q=a * min(1, k * 2): int(v * q)))
            y = self.y0 + i * self.step
            fr.paste(im, (int(self.cx - im.width / 2), int(y - im.height / 2)), im)


class InkWipe(El):
    """tinta menyebar menutupi layar lalu surut (transisi antarbagian)"""
    z = 100

    def __init__(self, t0, dur=1.0, col=(24, 46, 88), corner=(1, 1), **kw):
        super().__init__(t0, out=t0 + dur, **kw)
        self.dur, self.col = dur, col
        rng = np.random.default_rng(int(t0 * 10) + 3)
        small = rng.random((FH // 16, FW // 16))
        noise = np.asarray(Image.fromarray((small * 255).astype(np.uint8)).resize((FW, FH), Image.BICUBIC).filter(ImageFilter.GaussianBlur(18))) / 255.0
        fine = np.asarray(Image.fromarray((rng.random((FH // 4, FW // 4)) * 255).astype(np.uint8)).resize((FW, FH), Image.BICUBIC)) / 255.0
        yy, xx = np.mgrid[0:FH, 0:FW]
        cx, cy = corner[0] * FW, corner[1] * FH
        dist = np.sqrt((xx - cx) ** 2 + (yy - cy) ** 2) / math.hypot(FW, FH)
        self.field = dist * 0.75 + noise * 0.2 + fine * 0.05

    def register(self):
        if self.sfx_on:
            sfx(self.t0, 'whoosh', 0, 0.6)

    def draw(self, fr, t):
        u = (t - self.t0) / self.dur
        k = eio(clamp(u / 0.5)) if u < 0.5 else 1 - eio(clamp((u - 0.5) / 0.5))
        thr = k * 1.05
        m = (self.field < thr).astype(np.uint8) * 255
        if not m.any():
            return
        mask = Image.fromarray(m).filter(ImageFilter.GaussianBlur(1.2))
        fr.paste(Image.new('RGB', fr.size, self.col), (0, 0), mask)


class TypeScatter(El):
    """baris teks diketik huruf demi huruf dengan kursor, lalu huruf-hurufnya berhamburan dan hilang"""

    def __init__(self, t0, lines, x0, y0, size=30, per=0.035, tscatter=None, color=NAVY, lh=48, **kw):
        super().__init__(t0, **kw)
        self.lines, self.x0, self.y0, self.size, self.per, self.color, self.lh = lines, x0, y0, size, per, color, lh
        self.chars = []
        f = font('kr', size)
        k = 0
        for li, ln in enumerate(lines):
            x = x0
            for ch in ln:
                self.chars.append((ch, x, y0 + li * lh, k))
                x += f.getlength(ch)
                k += 1
        self.n = k
        self.ts = tscatter or (t0 + k * per + 1.0)
        rng = np.random.default_rng(7)
        self.vel = [(rng.uniform(-260, 260), rng.uniform(-420, 120), rng.uniform(-300, 300)) for _ in self.chars]
        self.font = f

    def register(self):
        if self.sfx_on:
            for i in range(0, self.n, 3):
                sfx(self.t0 + i * self.per, 'tick', 0, 0.12)
            sfx(self.ts, 'swoosh', 0.6, 0.4)

    def draw(self, fr, t):
        a = self.fade(t)
        d = ImageDraw.Draw(fr)
        shown = int((t - self.t0) / self.per)
        sk = (t - self.ts)
        last = None
        for i, (ch, x, y, k) in enumerate(self.chars):
            if k >= shown:
                break
            if sk > 0:
                vx, vy, vr = self.vel[i]
                tt = min(sk, 1.2)
                x2, y2 = x + vx * tt, y + vy * tt + 500 * tt * tt
                al = clamp(1 - sk / 1.0) * a
                if al <= 0:
                    continue
                sp = Image.new('RGBA', (self.size * 2, self.size * 2), (0, 0, 0, 0))
                ImageDraw.Draw(sp).text((self.size // 2, self.size // 2), ch, font=self.font, fill=self.color + (int(255 * al),))
                sp = sp.rotate(vr * tt, resample=Image.BICUBIC)
                fr.paste(sp, (int(x2 - self.size // 2), int(y2 - self.size // 2)), sp)
            else:
                d.text((x, y), ch, font=self.font, fill=mix(PAPER, self.color, a))
            last = (x + self.font.getlength(ch), y)
        if sk <= 0 and last and int(t * 2) % 2 == 0:
            d.rectangle((last[0] + 3, last[1] + 4, last[0] + 15, last[1] + self.size + 4), fill=mix(PAPER, self.color, a))


class TornLayer(El):
    """lapisan kertas sobek (tepi bergerigi + garis putih) meluncur masuk dari samping"""

    def __init__(self, t0, ytop, ybot, col, side='right', jag=26, halftone=False, seed=1, **kw):
        super().__init__(t0, **kw)
        self.ytop, self.ybot, self.col, self.side, self.halftone = ytop, ybot, col, side, halftone
        rng = np.random.default_rng(seed)
        xs = np.linspace(-20, FW + 20, 18)
        self.top = [(x, ytop + rng.uniform(-jag, jag) + (math.sin(x / 90 + seed) * jag)) for x in xs]
        im = Image.new('RGBA', (FW, FH), (0, 0, 0, 0))
        d = ImageDraw.Draw(im)
        poly = self.top + [(FW + 20, ybot), (-20, ybot)]
        d.polygon([(x, y - 3) for x, y in self.top] + [(FW + 20, ybot), (-20, ybot)], fill=(250, 248, 240, 255))   # tepi putih sobekan
        d.polygon(poly, fill=col + (255,))
        if halftone:
            for y in range(int(ytop), int(ybot), 14):
                for x in range(0, FW, 14):
                    d.ellipse((x + 4, y + 4, x + 8, y + 8), fill=tuple(max(0, c - 28) for c in col) + (255,))
        self.im = im

    def register(self):
        if self.sfx_on:
            sfx(self.t0, 'paper', 0, 0.5)

    def draw(self, fr, t):
        k = eo(clamp((t - self.t0) / 0.7))
        if k <= 0:
            return
        dx = (1 - k) * FW * (1 if self.side == 'right' else -1)
        im = self.im
        if self.fade(t) < 1:
            im = im.copy(); im.putalpha(im.getchannel('A').point(lambda v: int(v * self.fade(t))))
        fr.paste(im, (int(dx), 0), im)


class HalftoneSun(El):
    """matahari dari titik-titik (halftone) yang muncul dari tengah ke luar"""

    def __init__(self, t0, cx, cy, r=90, col=(206, 70, 60), **kw):
        super().__init__(t0, **kw)
        self.cx, self.cy, self.r, self.col = cx, cy, r, col

    def register(self):
        if self.sfx_on:
            sfx(self.t0, 'rise', 0.8, 0.4)

    def draw(self, fr, t):
        k = eo(clamp((t - self.t0) / 1.0))
        a = self.fade(t)
        d = ImageDraw.Draw(fr)
        st = 11
        for y in range(-self.r, self.r + 1, st):
            for x in range(-self.r, self.r + 1, st):
                dd = math.hypot(x, y) / self.r
                if dd > 1 or dd > k * 1.05:
                    continue
                rr = 4.6 * (1 - dd * 0.55)
                d.ellipse((self.cx + x - rr, self.cy + y - rr, self.cx + x + rr, self.cy + y + rr), fill=mix(PAPER, self.col, a))


class Flock(El):
    """burung (garis) terbang melintas layar sambil mengepak"""

    def __init__(self, t0, n=3, y=380, dur=3.0, direction=-1, **kw):
        super().__init__(t0, **kw)
        self.n, self.y, self.dur, self.dir = n, y, dur, direction

    def draw(self, fr, t):
        u = (t - self.t0) / self.dur
        if u < 0 or u > 1:
            return
        d = ImageDraw.Draw(fr)
        for i in range(self.n):
            x = (1 - u) * (FW + 120) - 60 - i * 70 if self.dir < 0 else u * (FW + 120) - 60 + i * 70
            x = x + i * 40 * self.dir
            y = self.y + i * 26 + math.sin(u * 8 + i) * 10
            f = math.sin(t * 9 + i) * 10
            s = 18 - i * 2
            d.line([(x - s, y - f), (x, y), (x + s, y - f)], fill=NAVY, width=3, joint='curve')

class OneTake(El):
    """satu tarikan kamera turun tanpa potongan: langit > gunung > air terjun > kolam > sungai > sosok kecil (tanpa wajah)"""
    z = -50

    def __init__(self, t0, dur=6.0, tall=3200, **kw):
        super().__init__(t0, **kw)
        self.dur, self.tall = dur, tall
        im = Image.new('RGB', (FW, tall), (226, 234, 238))
        d = ImageDraw.Draw(im)
        for y in range(0, 700):                                                        # langit gradasi
            c = int(250 - y * 0.05)
            d.line([(0, y), (FW, y)], fill=(c - 14, c - 6, c))
        d.ellipse((470, 180, 610, 320), fill=(206, 76, 62))                            # matahari
        d.polygon([(-40, 900), (220, 420), (300, 470), (470, 300), (760, 900)], fill=(70, 96, 140))     # gunung
        d.polygon([(380, 380), (470, 300), (560, 380), (520, 400), (470, 370), (420, 400)], fill=(244, 244, 240))
        d.rectangle((0, 880, FW, 2300), fill=(92, 128, 84))                           # tebing hijau
        rng = np.random.default_rng(3)
        for _ in range(140):
            x, y = rng.uniform(0, FW), rng.uniform(900, 2250)
            d.arc((x - 26, y - 10, x + 26, y + 12), 200, 340, fill=(124, 160, 100), width=4)
        d.rectangle((300, 880, 420, 2300), fill=(70, 110, 180))                       # air terjun
        d.ellipse((180, 2240, 540, 2420), fill=(150, 196, 226))                        # kolam
        d.rectangle((0, 2320, FW, tall), fill=(176, 196, 120))                         # tanah
        pts = [(360 + 140 * math.sin(i / 9), 2330 + i * 22) for i in range(40)]       # sungai berkelok
        for (x, y) in pts:
            d.ellipse((x - 90, y - 30, x + 90, y + 30), fill=(80, 126, 190))
        d.polygon([(0, tall - 420), (260, tall - 520), (FW, tall - 470), (FW, tall), (0, tall)], fill=(220, 196, 140))   # tepian pasir
        self.base = im
        self.fig = (470, tall - 520)

    def register(self):
        if self.sfx_on:
            sfx(self.t0, 'rise', min(2.5, self.dur * 0.4), 0.3)

    def draw(self, fr, t):
        u = eio(clamp((t - self.t0) / self.dur))
        y0 = int(u * (self.tall - fr.height))
        crop = self.base.crop((0, y0, FW, y0 + fr.height))
        d = ImageDraw.Draw(crop)
        ph = (t * 260) % 60                                                             # garis air jatuh bergerak
        for x in range(312, 412, 16):
            for yy in range(-60, fr.height, 60):
                ya = yy + ph + (x % 32)
                gy = ya + y0
                if 880 < gy < 2290:
                    d.line([(x, ya), (x, ya + 26)], fill=(236, 244, 250), width=4)
        for k in range(6):                                                              # buih di kaki air terjun
            bx, by = 300 + k * 22 + 8 * math.sin(t * 5 + k), 2280 - y0 + 6 * math.cos(t * 4 + k)
            d.ellipse((bx - 14, by - 12, bx + 14, by + 12), fill=(248, 250, 252), outline=(200, 214, 226))
        fx, fy = self.fig[0], self.fig[1] - y0                                         # sosok lidi kecil (tanpa wajah)
        d.ellipse((fx - 9, fy - 70, fx + 9, fy - 52), outline=NAVY, width=3)
        d.line([(fx, fy - 52), (fx, fy - 20), (fx - 12, fy), (fx, fy - 20), (fx + 12, fy)], fill=NAVY, width=3)
        d.line([(fx - 14, fy - 42), (fx + 14, fy - 42)], fill=NAVY, width=3)
        lx, ly = 360 + 120 * math.sin(t * 0.8), (t - self.t0) * 210 % fr.height        # daun merah mengikuti arus
        d.ellipse((lx - 7, ly - 4, lx + 7, ly + 4), fill=(200, 70, 50))
        fr.paste(crop, (0, 0))


class NightFest(El):
    """langit malam berbintang, untaian lentera berayun, kembang api meletup (latar penuh)"""
    z = -50

    def __init__(self, t0, fireworks=((1.0, 180, 300), (1.8, 520, 240), (2.6, 360, 380)), **kw):
        super().__init__(t0, **kw)
        self.fw = fireworks
        rng = np.random.default_rng(11)
        self.stars = [(rng.uniform(0, FW), rng.uniform(0, FH * 0.6), rng.uniform(1, 2.6), rng.uniform(0, 6)) for _ in range(90)]

    def register(self):
        if self.sfx_on:
            for (dt, x, y) in self.fw:
                sfx(self.t0 + dt, 'thud', 0, 0.25); sfx(self.t0 + dt + 0.05, 'ding', 0, 0.2)

    def draw(self, fr, t):
        a = self.fade(t)
        bg = Image.new('RGB', fr.size, (20, 28, 56))
        d = ImageDraw.Draw(bg)
        for y in range(0, fr.height, 8):
            c = int(18 + 30 * y / fr.height)
            d.line([(0, y), (fr.width, y)], fill=(c, c + 8, c + 34))
        for (x, y, r, ph) in self.stars:
            b = 0.5 + 0.5 * math.sin(t * 2 + ph)
            col = (int(200 + 55 * b), int(200 + 40 * b), 170)
            d.ellipse((x - r, y - r, x + r, y + r), fill=col)
        sw = math.sin(t * 1.6) * 6                                                      # untaian lentera
        pts = [(x, 170 + 40 * math.sin(x / FW * math.pi) + sw * math.sin(x / 120)) for x in range(-20, FW + 40, 20)]
        d.line(pts, fill=(60, 60, 70), width=2)
        for i, x in enumerate(range(40, FW, 80)):
            y = 170 + 40 * math.sin(x / FW * math.pi) + sw * math.sin(x / 120) + 18
            ox = math.sin(t * 2 + i) * 4
            d.line([(x, y - 18), (x + ox, y)], fill=(60, 60, 70), width=2)
            d.ellipse((x + ox - 22, y, x + ox + 22, y + 36), fill=(214, 82, 60))
            d.ellipse((x + ox - 10, y + 8, x + ox + 10, y + 28), fill=(250, 196, 120))
        for (dt, x, y) in self.fw:                                                     # kembang api
            k = (t - self.t0 - dt) / 1.4
            if 0 < k < 1:
                r = 20 + 140 * eo(k)
                al = 1 - k
                for j in range(18):
                    ang = j / 18 * math.tau
                    x1, y1 = x + math.cos(ang) * r * 0.55, y + math.sin(ang) * r * 0.55 + 30 * k * k
                    x2, y2 = x + math.cos(ang) * r, y + math.sin(ang) * r + 40 * k * k
                    col = (int(250 * al + 20), int(210 * al + 30), int(110 * al + 50))
                    d.line([(x1, y1), (x2, y2)], fill=col, width=3)
        d.rectangle((0, int(fr.height * 0.8), fr.width, fr.height), fill=(32, 34, 46))  # siluet atap
        for x in range(0, fr.width, 160):
            d.polygon([(x, int(fr.height * 0.8)), (x + 80, int(fr.height * 0.74)), (x + 160, int(fr.height * 0.8))], fill=(32, 34, 46))
        if a < 1:
            bg = Image.blend(fr.convert('RGB'), bg, a)
        fr.paste(bg)


class Iris(El):
    """lingkaran menutup ke satu titik lalu membuka lagi (transisi lensa)"""
    z = 100

    def __init__(self, t0, dur=1.2, cx=360, cy=640, col=(14, 61, 46), **kw):
        super().__init__(t0, out=t0 + dur, **kw)
        self.dur, self.cx, self.cy, self.col = dur, cx, cy + 72, col

    def register(self):
        if self.sfx_on:
            sfx(self.t0, 'swoosh', 0.5, 0.4); sfx(self.t0 + self.dur / 2, 'pop', 0, 0.4)

    def draw(self, fr, t):
        u = (t - self.t0) / self.dur
        k = 1 - eio(clamp(u / 0.5)) if u < 0.5 else eio(clamp((u - 0.5) / 0.5))
        R = k * math.hypot(fr.width, fr.height) * 0.62
        m = Image.new('L', fr.size, 255)
        ImageDraw.Draw(m).ellipse((self.cx - R, self.cy - R, self.cx + R, self.cy + R), fill=0)
        fr.paste(Image.new('RGB', fr.size, self.col), (0, 0), m)
        if R > 4:
            ImageDraw.Draw(fr).ellipse((self.cx - R, self.cy - R, self.cx + R, self.cy + R), outline=GOLDL, width=4)


class Ambient(El):
    """kabut tipis melayang + daun/kelopak berguguran (lapisan suasana, tidak mengganggu teks)"""
    z = 40

    def __init__(self, t0, leaves=10, mist=True, leaf_col=(200, 90, 60), **kw):
        super().__init__(t0, **kw)
        rng = np.random.default_rng(int(t0) + 5)
        self.leaves = [(rng.uniform(0, FW), rng.uniform(0, 1), rng.uniform(40, 90), rng.uniform(0, 6)) for _ in range(leaves)]
        self.mist, self.col = mist, leaf_col

    def draw(self, fr, t):
        a = self.fade(t) * clamp((t - self.t0) / 0.8)
        if a <= 0:
            return
        if self.mist:
            L = Image.new('RGBA', fr.size, (255, 255, 255, 0))
            d = ImageDraw.Draw(L)
            for i, y in enumerate((280, 560, 860)):
                x = ((t * (14 + i * 6)) % (FW + 700)) - 700
                d.ellipse((x, y - 40, x + 620, y + 40), fill=(255, 255, 255, int(46 * a)))
                d.ellipse((x - 820, y + 30, x - 320, y + 80), fill=(255, 255, 255, int(36 * a)))
            L = L.filter(ImageFilter.GaussianBlur(28))
            fr.paste(L, (0, 0), L)
        d = ImageDraw.Draw(fr)
        for (x0, y0, sp, ph) in self.leaves:
            y = ((y0 * FH + (t - self.t0) * sp) % (FH + 40)) - 20
            x = x0 + 30 * math.sin(t * 1.3 + ph)
            ang = t * 2 + ph
            dx, dy = 9 * math.cos(ang), 4 * math.sin(ang) + 3
            d.ellipse((x - abs(dx) - 2, y - abs(dy) - 2, x + abs(dx) + 2, y + abs(dy) + 2), fill=mix(PAPER, self.col, a * 0.9))


class ZoomThrough(El):
    """kamera menembus masuk ke satu titik (mis. kartu) sampai layar penuh, lalu memudar ke adegan berikutnya"""
    z = 60

    def __init__(self, t0, cx, cy, dur=1.0, **kw):
        super().__init__(t0, out=t0 + dur, **kw)
        self.cx, self.cy, self.dur = cx, cy + 72, dur

    def register(self):
        if self.sfx_on:
            sfx(self.t0, 'whoosh', 0, 0.6)

    def draw(self, fr, t):
        u = clamp((t - self.t0) / self.dur)
        z = 1 + 9 * eio(u) ** 1.6
        w2, h2 = fr.width / z, fr.height / z
        x0 = min(max(self.cx - w2 / 2, 0), fr.width - w2); y0 = min(max(self.cy - h2 / 2, 0), fr.height - h2)
        im = fr.transform(fr.size, Image.EXTENT, (x0, y0, x0 + w2, y0 + h2), Image.BICUBIC)
        if u > 0.6:
            im = Image.blend(im, Image.new('RGB', fr.size, (250, 248, 240)), (u - 0.6) / 0.4)
        fr.paste(im)


class Scroll(El):
    """gulungan kertas terbuka dari tengah ke kiri-kanan dan memperlihatkan isinya (elemen anak)"""

    def __init__(self, t0, cx, cy, w, h, children=(), dur=0.9, **kw):
        super().__init__(t0, **kw)
        self.cx, self.cy, self.w, self.h, self.ch, self.dur = cx, cy, w, h, list(children), dur

    def register(self):
        if self.sfx_on:
            sfx(self.t0, 'paper', 0, 0.6)
        for c in self.ch:
            c.out = self.out
            c.register()

    def draw(self, fr, t):
        a = self.fade(t)
        k = eio(clamp((t - self.t0) / self.dur))
        half = 12 + (self.w / 2 - 12) * k
        x0, x1 = self.cx - half, self.cx + half
        y0, y1 = self.cy - self.h / 2, self.cy + self.h / 2
        d = ImageDraw.Draw(fr)
        d.rectangle((x0, y0, x1, y1), fill=mix(PAPER, (246, 236, 208), a), outline=mix(PAPER, (196, 170, 120), a), width=2)
        L = Image.new('RGBA', fr.size, (0, 0, 0, 0))
        for c in self.ch:
            c.out, c.dim = self.out, self.dim
            if c.active(t):
                c.draw(L, t)
        clip = L.crop((int(x0) + 6, int(y0), int(x1) - 6, int(y1)))
        fr.paste(clip, (int(x0) + 6, int(y0)), clip)
        for x in (x0, x1):                                                              # batang penggulung
            d.rounded_rectangle((x - 12, y0 - 20, x + 12, y1 + 20), radius=10, fill=mix(PAPER, WOOD, a))
            d.ellipse((x - 16, y0 - 30, x + 16, y0 - 14), fill=mix(PAPER, GOLD, a)); d.ellipse((x - 16, y1 + 14, x + 16, y1 + 30), fill=mix(PAPER, GOLD, a))
