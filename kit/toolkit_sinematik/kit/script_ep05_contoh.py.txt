from engine import *

import unicodedata as _ud


def _n(s):
    return ''.join(c for c in s if _ud.category(c) != 'Mn').lower()


def W(t, kw, nth=0, end=False):
    i = seg_at(t)
    hits = [w for w in WORDS[i] if _n(kw) in _n(w[0])]
    if not hits:
        raise ValueError(f'kw {kw!r} not in seg {SEGS[i]}')
    w = hits[min(nth, len(hits) - 1)]
    return w[2] if end else w[1]

ELS = []
SCENES = []
CUR = None
TOTAL = 438.1


def scene(t0, t1, wh=True):
    global CUR
    CUR = {'t0': t0, 't1': t1, 'els': []}
    SCENES.append(CUR)
    if wh:
        sfx(t0, 'whoosh', 0, 0.55)


def A(el):
    if el.out is None:
        el.out = CUR['t1']
    CUR['els'].append(el)
    ELS.append(el)
    el.register()
    return el


def AA(els):
    return [A(e) for e in els]


def DIM(t, *keep):
    for el in CUR['els']:
        if el.dim is None and el.t0 < t and el not in keep:
            el.dim = t


def CLR(t):
    for el in CUR['els']:
        if el.out is None or el.out > t:
            el.out = t


def rasul(t, cx, cy, size=26, color=NAVY, pre='', extra=''):
    txt = pre + 'Rasulullah'
    w1 = tsprite(txt, 'kb', size, color).w
    w2 = tsprite('ﷺ', 'ar', int(size * 1.25), color).w
    w3 = tsprite(extra, 'kb', size, color).w + 6 if extra else 0
    left = cx - (w1 + 4 + w2 + w3) / 2
    A(Txt(t, txt, left, cy, size, color=color, ax='l'))
    A(Txt(t + 0.3, 'ﷺ', left + w1 + 4, cy + 2, int(size * 1.25), fn='ar', color=color, ax='l', mode='pop'))
    if extra:
        A(Txt(t + 0.45, extra, left + w1 + 4 + w2 + 6, cy, size, color=color, ax='l'))
