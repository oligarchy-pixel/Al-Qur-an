def fat_rows(items):

    for n, ic, isi, lafaz in items:

<div class="num" style="width:7mm;height:7mm;font-size:9pt">{n}</div>
<div style="flex:1"><p style="margin:0;font-size:8.6pt">{isi}</p>
<div class="ar" style="font-size:12pt;line-height:1.35;text-align:right">{esc(lafaz)}</div></div></div>'''
    return out
