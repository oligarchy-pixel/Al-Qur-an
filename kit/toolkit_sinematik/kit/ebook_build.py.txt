def esc(s):
    return html.escape(s, quote=False)


def ar(t, size=None):
    st = f' style="font-size:{size}pt"' if size else ''
    return f'<div class="ar"{st}>{esc(t)}</div>'


def dalil(arabic, tr, src=None, size=None):
    s = f'<div class="sticky">{esc(src)}</div>' if src else ''
    arr = arabic if isinstance(arabic, list) else [arabic]
    a = ''.join(ar(x, size) for x in arr)
    return f'<div class="dalil">{s}{a}<div class="tr">{tr}</div></div>'


def icon(n, cls='ico'):
    return f'<img class="{cls}" src="file://{HERE}/icons/{n}.png">'


def card(title, body, kind=''):
    return f'<div class="card {kind}"><h4>{title}</h4>{body}</div>'


def ck(items, x=False):
    return '<ul class="ck%s">' % (' x' if x else '') + ''.join(f'<li>{i}</li>' for i in items) + '</ul>'


def page(inner, label, no=None, cls=''):
    f = f'<div class="folio"><b>{esc(label)}</b>{" · " + str(no) if no else ""}</div>'
    return f'<section class="page {cls}">{inner}{f}</section>'


def chead(n, title, ic=None):
    return f'<div class="chead"><div class="num">{n}</div><h2 style="margin:0;flex:1">{title}</h2>{icon(ic, "icoS") if ic else ""}</div>'


async def to_pdf(html_path, pdf_path):
    async with async_playwright() as p:
        b = await p.chromium.launch(executable_path='/opt/pw-browsers/chromium') if __import__('os').path.exists('/opt/pw-browsers/chromium') else await p.chromium.launch()
        pg = await b.new_page()
        await pg.goto('file://' + html_path)
        await pg.wait_for_timeout(600)
        await pg.pdf(path=pdf_path, width='148mm', height='210mm', print_background=True, prefer_css_page_size=True)
        await b.close()


def build(mod_name, out_pdf):
    m = importlib.import_module(mod_name)
    pages = m.pages()
    doc = f'<!doctype html><html lang="id"><head><meta charset="utf-8"><title>{esc(m.TITLE)}</title><style>{CSS.replace("%(fd)s", FD)}</style></head><body>' + ''.join(pages) + '</body></html>'
    hp = os.path.join(HERE, mod_name + '.html')
    open(hp, 'w').write(doc)
    asyncio.run(to_pdf(hp, out_pdf))
    print(out_pdf)


if __name__ == '__main__':
    build(sys.argv[1], sys.argv[2])
